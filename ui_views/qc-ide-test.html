3 · The code under test

const IDE_API_URL = 'http://127.0.0.1:47823';
const DESKTOP_URL = 'quadcode://open';

// 1) Is an IDE listening? A refused connection is the only reliable signal.
async function isIdeRunning(timeoutMs = 1500) {
  try {
    const ctl = new AbortController();
    const timer = setTimeout(() => ctl.abort(), timeoutMs);
    const resp = await fetch(IDE_API_URL + '/ping', { method: 'GET', signal: ctl.signal });
    clearTimeout(timer);
    if (!resp.ok) return false;
    const data = await resp.json();
    return !!data && data.app === 'quadcodeai-ide';
  } catch (e) {
    return false;
  }
}

// 2) Hand the project over. 202 { status: "accepted" } — the IDE downloads the zip itself.
async function openProjectInIde(name, downloadUrl) {
  const resp = await fetch(IDE_API_URL + '/open-project', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ name: name, download_url: downloadUrl })
  });
  if (!resp.ok) throw new Error('IDE returned HTTP ' + resp.status);
  return resp.json();
}

// 3) No IDE running -> launch the desktop app via the custom scheme.
function launchProtocol(url) {
  const a = document.createElement('a');
  a.href = url; a.rel = 'noopener'; a.style.display = 'none';
  document.body.appendChild(a); a.click(); a.remove();
  const iframe = document.createElement('iframe');
  iframe.style.cssText = 'position:fixed;width:0;height:0;border:0;opacity:0';
  iframe.src = url; document.body.appendChild(iframe);
  setTimeout(() => iframe.remove(), 2500);
}

4 · Known behaviour to account for

  • Only the first launched IDE binds 127.0.0.1:47823. A second instance logs [Errno 48] Address already in use and stays silent; close the first one and the next launched instance takes the port.
  • A stale process holding the port makes /ping fail while the app looks alive — the page reports that as not running, which is correct but easy to misread.
  • quadcode://open needs handler support on the desktop side. Until that release ships, the deep-link button will do nothing visible — that is expected, not a page bug.
  • Mixed content: this page is HTTPS calling http://127.0.0.1. Chromium treats loopback as a secure context, so it is allowed; Safari may block it.
